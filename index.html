-- Table des administrateurs create table if not exists public.admin_users ( user_id uuid primary key references auth.users(id) on delete cascade, created_at timestamptz default now() ); alter table public.admin_users enable row level security; grant select on public.admin_users to authenticated; drop policy if exists "admin_read_own_status" on public.admin_users; create policy "admin_read_own_status" on public.admin_users for select to authenticated using (auth.uid() = user_id); -- Les admins gèrent toutes les annonces drop policy if exists "admin_manage_annonces" on public.annonces; create policy "admin_manage_annonces" on public.annonces for all to authenticated using (exists (select 1 from public.admin_users where user_id = auth.uid())) with check (exists (select 1 from public.admin_users where user_id = auth.uid())); -- Les admins gèrent toutes les demandes drop policy if exists "admin_manage_demandes" on public.demandes; create policy "admin_manage_demandes" on public.demandes for all to authenticated using (exists (select 1 from public.admin_users where user_id = auth.uid())) with check (exists (select 1 from public.admin_users where user_id = auth.uid())); -- Validation des données (nouvelles lignes uniquement) alter table public.annonces drop constraint if exists annonces_valide; alter table public.annonces add constraint annonces_valide check ( char_length(coalesce(description,'')) <= 500 and char_length(quartier) <= 50 and char_length(type_logement) <= 50 and telephone ~ '^\+?[0-9]{9,13}$' ) not valid; alter table public.demandes drop constraint if exists demandes_valide; alter table public.demandes add constraint demandes_valide check ( char_length(coalesce(description,'')) <= 500 and char_length(quartier) <= 50 and char_length(type_logement) <= 50 and telephone ~ '^\+?[0-9]{9,13}$' ) not valid;
